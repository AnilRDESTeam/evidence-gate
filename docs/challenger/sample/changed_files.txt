=== app/config.py ===
# Shop settings
FREE_SHIPPING_THRESHOLD = 75.00
STANDARD_SHIPPING = 5.99

=== app/pricing/shipping.py ===
from app.config import FREE_SHIPPING_THRESHOLD, STANDARD_SHIPPING

def shipping_for(subtotal: float) -> float:
    # Free shipping at $75 or more
    if subtotal >= FREE_SHIPPING_THRESHOLD:
        return 0.00
    return STANDARD_SHIPPING
